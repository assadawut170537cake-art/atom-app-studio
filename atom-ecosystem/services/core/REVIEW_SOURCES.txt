class AudioRecorder {

    }
    private val running = AtomicBoolean(false)
    private var audioRecord: AudioRecord? = null
    private val bufferSize: Int by lazy {
        val minBufferSize = AudioRecord.getMinBufferSize(
            SAMPLE_RATE,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT
        )
        (minBufferSize * 2).coerceAtLeast(CHUNK_SIZE_BYTES * 2)
    }
    @Synchronized
    fun start(): Boolean {
        if (running.get()) return true
        return try {
            val recorder = AudioRecord(
                MediaRecorder.AudioSource.VOICE_RECOGNITION,
                SAMPLE_RATE,
                AudioFormat.CHANNEL_IN_MONO,
                AudioFormat.ENCODING_PCM_16BIT,
                bufferSize
            )
            if (recorder.state != AudioRecord.STATE_INITIALIZED) {
                recorder.release()
                return false
            }
            recorder.startRecording()
            audioRecord = recorder
            running.set(true)
            true
        } catch (_: SecurityException) { false }
        catch (_: Exception) { false }
    }
    fun readChunk(): ByteArray? {
        if (!running.get()) return null
        val recorder = audioRecord ?: return null
        val chunk = ByteArray(CHUNK_SIZE_BYTES)
        var offset = 0
        while (offset < chunk.size && running.get()) {
            val read = recorder.read(
                chunk, offset, chunk.size - offset,
                AudioRecord.READ_BLOCKING
            )
            if (read <= 0) return null
            offset += read
        }
        return if (offset == chunk.size && running.get()) chunk else null
    }

        audioRecord = null
    }

import java.util.concurrent.TimeUnit
object AtomApiClient {
    private val authInterceptor = Interceptor { chain ->
        val request = chain.request()
            .newBuilder()
            .addHeader("X-Atom-Secret", BuildConfig.ATOM_SECRET)
            .build()
        chain.proceed(request)
    }
    private val loggingInterceptor = HttpLoggingInterceptor().apply {
        level = HttpLoggingInterceptor.Level.BASIC
    }
    private val client = OkHttpClient.Builder()
        .addInterceptor(authInterceptor)
        .addInterceptor(loggingInterceptor)
        .connectTimeout(10, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .writeTimeout(30, TimeUnit.SECONDS)
        .build()
    val api: FridayApiService = Retrofit.Builder()
        .baseUrl(BuildConfig.BASE_URL + "/")
        .client(client)
        .addConverterFactory(GsonConverterFactory.create())
        .build()
        .create(FridayApiService::class.java)
}

import java.util.concurrent.TimeUnit
object AtomApiClient {
    private val authInterceptor = Interceptor { chain ->
        val request = chain.request()
            .newBuilder()
            .addHeader("X-Atom-Secret", BuildConfig.ATOM_SECRET)
            .build()
        chain.proceed(request)
    }
    private val loggingInterceptor = HttpLoggingInterceptor().apply {
        level = HttpLoggingInterceptor.Level.BODY
    }
    private val client = OkHttpClient.Builder()
        .addInterceptor(authInterceptor)
        .addInterceptor(loggingInterceptor)
        .connectTimeout(10, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .writeTimeout(30, TimeUnit.SECONDS)
        .build()
    val api: UltronApiService = Retrofit.Builder()
        .baseUrl(BuildConfig.BASE_URL + "/")
        .client(client)
        .addConverterFactory(GsonConverterFactory.create())
        .build()
        .create(UltronApiService::class.java)
}

NEW_SERVICE="atom-core.service"
echo "=== ATOM Migration Start ==="

if systemctl list-unit-files | grep -q "$OLD_SERVICE"; then
    systemctl stop jarvis-core || true
    systemctl disable jarvis-core || true

echo "[3] Create atom system user"
if ! id atom >/dev/null 2>&1; then
    useradd --system --create-home --shell /usr/sbin/nologin atom
fi
echo "[4] Create directories"
mkdir -p "$NEW_DIR/data"
mkdir -p "$NEW_DIR/secrets"

chown -R atom:atom "$NEW_DIR"
